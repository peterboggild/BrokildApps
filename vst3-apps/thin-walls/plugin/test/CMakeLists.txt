cmake_minimum_required(VERSION 3.22)
project(ThinWallsTest LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

# The bench measures numbers that get quoted (RT60, DRR, ITD), so no fast-math.
add_executable(twtest bench.cpp ../Source/Engine.cpp)
target_include_directories(twtest PRIVATE ../Source)

if(MSVC)
    target_compile_options(twtest PRIVATE /O2 /fp:precise /W4 /bigobj)
else()
    target_compile_options(twtest PRIVATE -O2 -fno-fast-math -Wall -Wextra)
endif()

add_executable(twprobe probe.cpp ../Source/Engine.cpp)
target_include_directories(twprobe PRIVATE ../Source)
if(MSVC)
    target_compile_options(twprobe PRIVATE /O2 /fp:precise /bigobj)
endif()

add_executable(twquality quality.cpp ../Source/Engine.cpp)
target_include_directories(twquality PRIVATE ../Source)
if(MSVC)
    target_compile_options(twquality PRIVATE /O2 /fp:precise /bigobj)
    target_link_options(twquality PRIVATE /STACK:33554432)
else()
    target_compile_options(twquality PRIVATE -O2 -fno-fast-math)
endif()

add_executable(twdoorwalk doorwalk.cpp ../Source/Engine.cpp)
target_include_directories(twdoorwalk PRIVATE ../Source)
if(MSVC)
    target_compile_options(twdoorwalk PRIVATE /O2 /fp:precise /bigobj)
    target_link_options(twdoorwalk PRIVATE /STACK:33554432)
endif()

add_executable(twfurn furniture.cpp ../Source/Engine.cpp)
target_include_directories(twfurn PRIVATE ../Source)
if(MSVC)
    target_compile_options(twfurn PRIVATE /O2 /fp:precise /bigobj)
    target_link_options(twfurn PRIVATE /STACK:33554432)
endif()

include(../cmake/sofa.cmake)
add_executable(twrender render.cpp ../Source/Engine.cpp ../Source/HrtfSofa.cpp ../Source/OfflineRender.cpp ../Source/LateRays.cpp ../Source/Fdtd.cpp)
target_include_directories(twrender PRIVATE ../Source)
target_link_libraries(twrender PRIVATE tw_mysofa)
target_compile_definitions(twrender PRIVATE TW_SOFA_KEMAR="${TW_SOFA_KEMAR}")
if(MSVC)
    target_compile_options(twrender PRIVATE /O2 /fp:precise /bigobj)
    target_link_options(twrender PRIVATE /STACK:33554432)
endif()
