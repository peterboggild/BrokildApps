cmake_minimum_required(VERSION 3.22)
project(MagnetoTransportTest LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

# The bench measures numbers that get quoted, so no fast-math.
add_executable(mttest bench.cpp ../Source/Engine.cpp ../Source/Playback.cpp ../Source/Equaliser.cpp ../Source/Transport.cpp ../Source/Stock.cpp ../Source/Medium.cpp)
target_include_directories(mttest PRIVATE ../Source)
# the plug-in's build id, for the tips numbers it writes (--tips=)
file(STRINGS ../CMakeLists.txt MT_BID_LINE REGEX "^set.MT_BUILD_ID")
string(REGEX MATCH "[0-9]+[.][0-9]+" MT_BID "${MT_BID_LINE}")
target_compile_definitions(mttest PRIVATE MT_BENCH_BUILD="${MT_BID}")
if(MSVC)
    target_compile_definitions(mttest PRIVATE _USE_MATH_DEFINES)
    target_compile_options(mttest PRIVATE /O2 /fp:precise /W4)
else()
    target_compile_options(mttest PRIVATE -O2 -fno-fast-math -Wall -Wextra)
endif()
