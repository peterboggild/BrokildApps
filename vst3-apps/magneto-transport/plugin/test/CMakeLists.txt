cmake_minimum_required(VERSION 3.22)
project(MagnetoTransportTest LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

# The bench measures numbers that get quoted, so no fast-math.
add_executable(mttest bench.cpp ../Source/Engine.cpp ../Source/Playback.cpp ../Source/Equaliser.cpp ../Source/Transport.cpp ../Source/Stock.cpp ../Source/Medium.cpp)
target_include_directories(mttest PRIVATE ../Source)
if(MSVC)
    target_compile_definitions(mttest PRIVATE _USE_MATH_DEFINES)
    target_compile_options(mttest PRIVATE /O2 /fp:precise /W4)
else()
    target_compile_options(mttest PRIVATE -O2 -fno-fast-math -Wall -Wextra)
endif()

# Prints engine internals; asserts nothing. The tool that settles a question
# the bench raises.
add_executable(mtprobe probe.cpp ../Source/Engine.cpp ../Source/Playback.cpp ../Source/Equaliser.cpp ../Source/Transport.cpp ../Source/Stock.cpp ../Source/Medium.cpp)
target_include_directories(mtprobe PRIVATE ../Source)
if(MSVC)
    target_compile_definitions(mtprobe PRIVATE _USE_MATH_DEFINES)
    target_compile_options(mtprobe PRIVATE /O2 /fp:precise)
endif()

# where the time goes, part by part (prints, asserts nothing)
add_executable(mtcost cost.cpp ../Source/Engine.cpp ../Source/Playback.cpp ../Source/Equaliser.cpp ../Source/Transport.cpp ../Source/Stock.cpp ../Source/Medium.cpp)
target_compile_features(mtcost PRIVATE cxx_std_17)
if (MSVC)
  target_compile_options(mtcost PRIVATE /O2 /fp:precise)
  target_include_directories(mtcost PRIVATE ../Source)
  target_compile_definitions(mtcost PRIVATE _USE_MATH_DEFINES)
  target_link_options(mtcost PRIVATE /STACK:33554432)
endif()

# the equaliser fits per coating (prints)
add_executable(mteqfit eqfit.cpp ../Source/Engine.cpp ../Source/Playback.cpp ../Source/Equaliser.cpp ../Source/Transport.cpp ../Source/Stock.cpp ../Source/Medium.cpp)
target_include_directories(mteqfit PRIVATE ../Source)
if (MSVC)
  target_compile_definitions(mteqfit PRIVATE _USE_MATH_DEFINES)
  target_compile_options(mteqfit PRIVATE /O2 /fp:precise)
endif()

# dropouts, counted from the medium and through the engine (prints)
add_executable(mtdrop dropprobe.cpp ../Source/Engine.cpp ../Source/Playback.cpp ../Source/Equaliser.cpp ../Source/Transport.cpp ../Source/Stock.cpp ../Source/Medium.cpp)
target_include_directories(mtdrop PRIVATE ../Source)
if (MSVC)
  target_compile_definitions(mtdrop PRIVATE _USE_MATH_DEFINES)
  target_compile_options(mtdrop PRIVATE /O2 /fp:precise)
endif()
